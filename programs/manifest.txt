# ROM slots for tools/gen_rom.py -> rtl/rom.v: `slot file`, files in this directory.
# Slot numbers are permanent: a new program takes an unused slot, an old slot never
# shifts, so a host that knows a number keeps working. Slot 0 is NONE, the core halts.
# 4-bit select: slots 1..15 are valid, 12..15 unused; an unused slot reads as slot 0.
1   uart_tx_0x55.asm
2   uart_tx_pull.asm
3   uart_tx_loop.asm
4   uart_rx.asm
5   spi_tx_lsb.asm
6   spi_tx_msb.asm
7   spi_duplex_lsb.asm
8   spi_duplex_msb.asm
9   i2c_write.asm
10  i2c_write_stretch.asm
11  i2c_write_addr_data.asm
